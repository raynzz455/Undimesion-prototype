   339→// DYNAMIC LIST — reusable for work history, education, skills
   340→// ═══════════════════════════════════════════════════════════════════════════
   341→function DynamicList<T extends Record<string, unknown>>({
   342→  title, items, setItems, fields,
   343→}: {
   344→  title: string;
   345→  items: T[];
   346→  setItems: (items: T[]) => void;
   347→  fields: { key: string; placeholder: string; width?: string; textarea?: boolean; type?: string }[];
   348→}) {
   349→  const update = (i: number, key: string, v: string | number | boolean) => setItems(items.map((item, idx) => (idx === i ? { ...item, [key]: v } : item)));
   350→  const add = () => { const newItem = {} as T; fields.forEach((f) => { (newItem as Record<string, unknown>)[f.key] = f.type === "number" ? 0 : ""; }); setItems([...items, newItem]); };
   351→  const remove = (i: number) => setItems(items.filter((_, idx) => idx !== i));
   352→
   353→  return (
   354→    <div className="mb-3">
   355→      <div className="flex items-center justify-between mb-2">
   356→        <span className="font-mono-ud text-xs font-bold text-[#d4ff00] tracking-wider">{title} ({items.length})</span>
   357→        <button onClick={add} className="flex items-center gap-1 bg-[#d4ff00] text-black font-mono-ud text-xs font-black px-2 py-1 border-2 border-[#d4ff00] hover:bg-transparent hover:text-[#d4ff00] transition-colors no-color-transition min-h-[36px]"><Plus className="w-3 h-3" /> ADD</button>
   358→      </div>
   359→      <div className="space-y-2">
   360→        {items.map((item, i) => (
   361→          <div key={i} className="border-2 border-[#d4ff00]/30 p-2 bg-[#1a1a1a] relative">
   362→            <div className="grid grid-cols-1 md:grid-cols-6 gap-2">
   363→              {fields.map((f) => (
   364→                <div key={f.key} className={f.width || ""}>
   365→                  {f.textarea ? (
   366→                    <textarea value={String(item[f.key] ?? "")} onChange={(e) => update(i, f.key, e.target.value)} placeholder={f.placeholder} rows={1} className="w-full px-2 py-1.5 bg-black border border-[#d4ff00]/30 text-white font-mono-ud text-xs resize-none focus:outline-none focus:border-[#ff00ff]" />
   367→                  ) : (
   368→                    <input value={String(item[f.key] ?? "")} onChange={(e) => update(i, f.key, f.type === "number" ? Number(e.target.value) : e.target.value)} placeholder={f.placeholder} type={f.type || "text"} className="w-full px-2 py-1.5 bg-black border border-[#d4ff00]/30 text-white font-mono-ud text-xs focus:outline-none focus:border-[#ff00ff]" />
   369→                  )}
   370→                </div>
   371→              ))}
   372→            </div>
   373→            <button onClick={() => remove(i)} className="absolute top-1 right-1 p-1.5 bg-[#ff4d4d]/80 hover:bg-[#ff4d4d] text-white border border-white/30 transition-colors no-color-transition" title="Remove"><Trash2 className="w-3 h-3" /></button>
   374→          </div>
   375→        ))}
   376→      </div>
   377→    </div>
   378→  );
   379→}

   924→          <textarea value={description} onChange={(e) => setDescription(e.target.value.slice(0, 500))} placeholder="DESCRIPTION * (max 500)" rows={3} className="block w-full px-3 py-2 border-2 border-[#d4ff00] bg-[#1a1a1a] text-white font-mono-ud text-sm resize-none focus:outline-none focus:border-[#ff00ff]" />
   925→          <div className="grid grid-cols-2 gap-2">
