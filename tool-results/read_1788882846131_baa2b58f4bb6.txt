   824→function AchievementsTab() {
   825→  const { data, refetch } = useFetch<{ achievements: DbAchievement[]; count: number }>("/api/achievements");
   826→  const { chaosFetch } = useChaosFetch();
   827→  const { play } = useSfx();
   828→  const [title, setTitle] = useState("");
   829→  const [year, setYear] = useState(String(new Date().getFullYear()));
   830→  const [description, setDescription] = useState("");
   831→  const [memberId, setMemberId] = useState("aldi");
   832→  const [error, setError] = useState<string | null>(null);
   833→  const [success, setSuccess] = useState<string | null>(null);
   834→  const [deletingId, setDeletingId] = useState<string | null>(null);
   835→
   836→  const achievements = data?.achievements ?? [];
   837→
   838→  const handleCreate = async () => {
   839→    if (!title.trim() || !description.trim()) { setError("Title dan description wajib diisi."); return; }
   840→    setError(null); setSuccess(null);
   841→    try {
   842→      const res = await chaosFetch("/api/achievements", {
   843→        method: "POST",
   844→        body: JSON.stringify({ title, year, description, memberId }),
   845→      });
   846→      const d = await res.json();
   847→      if (!res.ok) throw new Error(d.error || "Gagal");
   848→      setSuccess(`✓ Achievement "${d.title}" dibuat!`);
   849→      setTitle(""); setDescription("");
   850→      refetch(); play("submit");
   851→      setTimeout(() => setSuccess(null), 4000);
   852→    } catch (e) { setError(e instanceof Error ? e.message : "Gagal"); }
   853→  };
   854→
   855→  const handleDelete = async (id: string, title: string) => {
   856→    if (!confirm(`Hapus achievement "${title}"?`)) return;
   857→    setDeletingId(id);
   858→    try {
   859→      const res = await chaosFetch(`/api/achievements?id=${id}`, { method: "DELETE" });
   860→      if (!res.ok) { const d = await res.json(); throw new Error(d.error); }
   861→      setSuccess(`✓ Achievement dihapus.`);
   862→      refetch(); play("close");
   863→      setTimeout(() => setSuccess(null), 3000);
   864→    } catch (e) { setError(e instanceof Error ? e.message : "Gagal"); }
   865→    finally { setDeletingId(null); }
   866→  };

   869→    <div className="space-y-4">
   870→      {/* Create form */}
   871→      <div className="border-8 border-[#d4ff00] bg-black p-6 shadow-[12px_12px_0_#d4ff00]">

   877→            placeholder="ACHIEVEMENT TITLE *"
   878→            className="block w-full px-3 py-2 border-2 border-[#d4ff00] bg-[#1a1a1a] text-white font-mono-ud text-sm focus:outline-none focus:border-[#ff00ff]" />

   880→            placeholder="DESCRIPTION * (max 500 chars)" rows={3}
   881→            className="block w-full px-3 py-2 border-2 border-[#d4ff00] bg-[#1a1a1a] text-white font-mono-ud text-sm resize-none focus:outline-none focus:border-[#ff00ff]" />

   883→            <input value={year} onChange={(e) => setYear(e.target.value)} placeholder="YEAR"
   884→              className="px-3 py-2 border-2 border-[#d4ff00] bg-[#1a1a1a] text-white font-mono-ud text-sm focus:outline-none focus:border-[#ff00ff]" />

   888→            </select>
   889→          </div>
   890→          {error && <div className="bg-[#ff4d4d] text-white px-3 py-2 border-2 border-white font-mono-ud text-xs font-bold">! {error}</div>}
   891→          {success && <div className="bg-[#00ff00] text-black px-3 py-2 border-2 border-black font-mono-ud text-xs font-bold">{success}</div>}
   892→          <button onClick={handleCreate}
