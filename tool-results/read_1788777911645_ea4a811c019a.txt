    26→    26→function ScrollProgress() {

    28→    28→  useEffect(() => {
    29→    29→    const handleScroll = () => {
    30→    30→      const scrollHeight =
    31→    31→        document.documentElement.scrollHeight -
    32→    32→        document.documentElement.clientHeight
    33→    33→      const currentScroll = document.documentElement.scrollTop
    34→    34→      setWidth((currentScroll / scrollHeight) * 100)
    35→    35→    }

    41→    41→      className="fixed top-0 left-0 h-2 bg-[#ff4d4d] z-[60] transition-all duration-75 no-color-transition"
