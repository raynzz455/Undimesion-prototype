   159→function AchievementImageCarousel({ images, color }: { images: string[]; color: string }) {
   160→  const [idx, setIdx] = useState(0);
   161→
   162→  const next = () => setIdx((p) => (p + 1) % images.length);
   163→  const prev = () => setIdx((p) => (p - 1 + images.length) % images.length);
   164→
   165→  return (
   166→    <div className="mb-4 border-4 border-black dark:border-white bg-black overflow-hidden relative">
   167→      <div className="font-mono-ud text-sm font-black tracking-[0.2em] uppercase text-white/60 px-3 py-1 border-b-2 border-white/20 flex items-center justify-between">
   168→        <span>▸ EVIDENCE / CERTIFICATE</span>
   169→        <span>{idx + 1}/{images.length}</span>
   170→      </div>
   171→      <div className="relative aspect-[4/3] overflow-hidden">
   172→        <img
   173→          src={images[idx]}
   174→          alt={`Achievement evidence ${idx + 1}`}
   175→          className="w-full h-full object-cover"
   176→          key={idx}
   177→        />
   178→        <div className="absolute inset-0 ud-scanlines opacity-20 pointer-events-none" />
   179→        {images.length > 1 && (
   180→          <>
   181→            <button
   182→              onClick={prev}
   183→              className="absolute left-1 top-1/2 -translate-y-1/2 w-8 h-8 flex items-center justify-center bg-black/70 border-2 border-white text-white hover:bg-white hover:text-black transition-colors"
   184→              aria-label="Previous image"
   185→            >
   186→              <ChevronLeft className="w-4 h-4" />
   187→            </button>
   188→            <button
   189→              onClick={next}
   190→              className="absolute right-1 top-1/2 -translate-y-1/2 w-8 h-8 flex items-center justify-center bg-black/70 border-2 border-white text-white hover:bg-white hover:text-black transition-colors"

   192→            >
   193→              <ChevronRight className="w-4 h-4" />
   194→            </button>
   195→          </>
   196→        )}
   197→      </div>
   198→      {/* Dots */}
   199→      {images.length > 1 && (
   200→        <div className="flex justify-center gap-1 py-1.5 bg-black">
   201→          {images.map((_, i) => (
   202→            <button
   203→              key={i}
   204→              onClick={() => setIdx(i)}
   205→              className={cn(
   206→                "w-2 h-2 border border-white transition-colors",
   207→                i === idx ? "bg-white" : "bg-transparent",
   208→              )}
   209→              aria-label={`Go to image ${i + 1}`}
   210→            />
   211→          ))}
   212→        </div>
   213→      )}
   214→    </div>
   215→  );

   218→function AchievementModal({
   219→  achievement,
   220→  memberColor,
   221→  onClose,
   222→}: {
   223→  achievement: Achievement | null;
   224→  memberColor: string;
   225→  onClose: () => void;
   226→}) {
   227→  const open = achievement !== null;
   228→  const panelRef = useRef<HTMLDivElement>(null);
   229→  useFocusTrap(panelRef, open);
   230→
   231→  useEffect(() => {
   232→    if (!open) return;
   233→    const onKey = (e: KeyboardEvent) => {
   234→      if (e.key === "Escape") {
   235→        document.body.classList.remove("modal-open");
   236→        onClose();
   237→      }
   238→    };
   239→    window.addEventListener("keydown", onKey);
   240→    return () => window.removeEventListener("keydown", onKey);
   241→  }, [open, onClose]);
   242→
   243→  // Hide navbar when modal is open
   244→  useEffect(() => {
   245→    if (open) {
   246→      document.body.classList.add("modal-open");
   247→    } else {
   248→      document.body.classList.remove("modal-open");
   249→    }
   250→    return () => { document.body.classList.remove("modal-open"); };
   251→  }, [open]);
   252→
   253→  return (
   254→    <AnimatePresence>
   255→      {open && achievement && (
   256→        <motion.div
   257→          className="fixed inset-0 z-[85] flex items-center justify-center p-4 md:p-8"
   258→          initial={{ opacity: 0 }}
   259→          animate={{ opacity: 1 }}
   260→          exit={{ opacity: 0 }}
   261→          transition={{ duration: 0.2 }}
   262→        >
   263→          <div className="absolute inset-0 bg-black/80 backdrop-blur-sm" onClick={onClose} />
   264→
   265→          {/* Relative wrapper */}
   266→          <div className="relative w-full max-w-md">
   267→            {/* X Close */}
   268→            <button
   269→              onClick={onClose}
   270→              className="absolute -top-4 -right-4 z-30 w-10 h-10 flex items-center justify-center bg-[#ff4d4d] text-white border-4 border-black dark:border-white shadow-[4px_4px_0_#000] dark:shadow-[4px_4px_0_#fff] hover:rotate-90 transition-transform no-color-transition"
   271→              aria-label="Close"
   272→            >
   273→              <X className="w-5 h-5" />
   274→            </button>
   275→
   276→          <motion.div
   277→            ref={panelRef}
   278→            role="dialog"
   279→            aria-modal="true"
   280→            aria-labelledby="ud-achievement-title"
   281→            tabIndex={-1}
   282→            className="relative w-full max-h-[90vh] overflow-y-auto border-8 border-black dark:border-white bg-white dark:bg-[#09090b] shadow-[8px_8px_0_#000] md:shadow-[16px_16px_0_#000] dark:md:shadow-[16px_16px_0_#d4ff00] outline-none"
   283→            initial={{ scale: 0.85, y: 30, rotate: -2 }}
   284→            animate={{ scale: 1, y: 0, rotate: 0 }}
   285→            exit={{ scale: 0.85, y: 30, rotate: -2 }}
   286→            transition={{ type: "spring", stiffness: 300, damping: 24 }}
   287→            onAnimationComplete={() => {
   288→              if (panelRef.current) panelRef.current.scrollTop = 0;
   289→            }}
   290→          >
   291→
   292→            {/* Colored header banner */}
   293→            <div className="p-6 border-b-8 border-black dark:border-white" style={{ backgroundColor: memberColor }}>
   294→              <div className="flex items-center justify-center mb-2">
   295→                <div className="w-16 h-16 flex items-center justify-center bg-black border-4 border-black">
   296→                  <Trophy className="w-8 h-8 text-white" />
   297→                </div>
   298→              </div>
   299→              <div className="font-mono-ud text-sm font-black tracking-[0.3em] uppercase text-black/60 text-center">
   300→                ▸ ACHIEVEMENT UNLOCKED
   301→              </div>
   302→            </div>
   303→
   304→            {/* Body */}
   305→            <div className="p-6">
   306→              <h3
   307→                id="ud-achievement-title"
   308→                className="font-bebas text-4xl md:text-5xl text-black dark:text-white leading-none mb-2 text-center"
   309→              >
   310→                {achievement.title}
   311→              </h3>
   312→              <div className="font-mono-ud text-xs font-black text-black/50 dark:text-white/50 tracking-widest mb-4 text-center">
   313→                ◆ {achievement.year} ◆
   314→              </div>
   315→              <div className="border-4 border-black dark:border-white bg-black dark:bg-white p-4 mb-4">
   316→                <p className="font-mono-ud text-sm text-white dark:text-black leading-relaxed text-center">
   317→                  {achievement.description}
   318→                </p>
   319→              </div>
   320→              {/* Achievement photo carousel (sertifikat, medali, foto kemenangan) */}
   321→              {achievement.images && achievement.images.length > 0 && (
   322→                <AchievementImageCarousel images={achievement.images} color={memberColor} />
